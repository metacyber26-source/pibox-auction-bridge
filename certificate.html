function updateCertificateUI(data) { const shortId = data.id ? data.id.substring(0, 6).toUpperCase() : 'NFT-GM'; document.getElementById('ref-number').innerText = `NFT-GM/AUC-CERT/2026/10/${shortId}`; document.getElementById('asset-name').innerText = data.title || 'Nusantara NFT Asset'; document.getElementById('tx-id').innerText = data.hash || '-'; // Format Waktu On-Chain (mengambil dari waktu pembuatan atau waktu kustom on-chain jika ada) const onchainTimeStr = data.onchain_time || data.created_at; const formattedOnchainTime = onchainTimeStr ? onchainTimeStr.replace('T', ' ').substring(0, 19) + ' UTC' : '-'; document.getElementById('onchain-time').innerText = formattedOnchainTime; // Format Waktu Penutupan Lelang (mengambil dari ends_at atau updated_at jika status finished) const closingTimeStr = data.closing_time || data.ends_at || data.updated_at || data.created_at; const formattedClosingTime = closingTimeStr ? closingTimeStr.replace('T', ' ').substring(0, 19) + ' UTC' : '-'; document.getElementById('closing-time').innerText = formattedClosingTime; document.getElementById('closing-price').innerText = `${data.highest_bid || data.reserve_price || '0'} Pi`; const seller = data.seller || 'seller_official'; document.getElementById('seller-username').innerText = seller.startsWith('@') ? seller : '@' + seller; const winner = data.highest_bidder || data.winner || 'Pioneer'; document.getElementById('winner-username').innerText = winner.startsWith('@') ? winner : '@' + winner; }
